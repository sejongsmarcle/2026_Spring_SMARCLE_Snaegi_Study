설정한 임계값 1 : 거리 100cm
설정한 임계값 1 : 거리 50cm

거리 100cm 초과할 때 : 초록불 동작
거리 50cm초과 100cm 이하일 때 : 노란불 동작
거리 50cm이하일 때 : 빨간불 동작

<초음파센서 / 코드>

int sensor = 8;
int green = 12;
int yellow = 11;
int red = 10;

int limit1 = 100;//임계거리값 1
int limit2 = 50;//임계거리값2

void setup() {
  pinMode(green, OUTPUT);
  pinMode(yellow, OUTPUT);
  pinMode(red, OUTPUT);

  Serial.begin(9600);
}

void loop() {
  long t;
  int distance;

  pinMode(sensor, OUTPUT);
  digitalWrite(sensor, LOW);
  delayMicroseconds(2);
  digitalWrite(sensor, HIGH);
  delayMicroseconds(5);
  digitalWrite(sensor, LOW);

  pinMode(sensor, INPUT);
  t = pulseIn(sensor, HIGH);
  distance = t * 0.034 / 2;
  Serial.print(distance);

  if (distance > limit1) {//초록
    digitalWrite(green, HIGH);
    digitalWrite(yellow, LOW);
    digitalWrite(red, LOW);
  }
  else if (distance > limit2) {//노랑
    digitalWrite(green, LOW);
    digitalWrite(yellow, HIGH);
    digitalWrite(red, LOW);
  }
  else {//빨강
    digitalWrite(green, LOW);
    digitalWrite(yellow, LOW);
    digitalWrite(red, HIGH);
  }




  

  delay(200);
}
