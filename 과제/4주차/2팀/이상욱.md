#include <Servo.h>
#include <LiquidCrystal.h>

LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

Servo myServo;

void setup() {
  myServo.attach(9);

  lcd.begin(16, 2);

  lcd.print("Servo Start");

  delay(1000);

  lcd.clear();
}

void loop() {

  for(int angle = 0; angle <= 180; angle += 30) {

    myServo.write(angle);

    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("Servo Angle");

    lcd.setCursor(0, 1);
    lcd.print(angle);
    lcd.print((char)223);
    lcd.print(" ");

    delay(500);
  }

서보모터는 Arduino의 디지털 9번 핀에 연결되어 PWM 신호를 통해 각도를 제어하며, LCD는 디지털 핀들과 연결되어 현재 각도를 사용자에게 표시한다. 또한 브레드보드의 전원 라인을 이용하여 LCD와 서보모터가 동일한 5V 전원을 공유하도록 구성하였다.
반복문을 통해 서보모터의 각도를 0도부터 180도까지 30도 간격으로 증가시킨다. 각도가 변경될 때마다 LCD에는 현재 각도가 출력되며, 0.5초 동안 유지된 후 다음 각도로 이동한다.
