```cpp
#include <Servo.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

void setup() {
  myServo.attach(9);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("Servo Angle:");
}

void loop() {

  for (int angle = 0; angle <= 180; angle += 30) {

    myServo.write(angle);

    lcd.setCursor(0, 1);
    lcd.print("    ");
    lcd.setCursor(0, 1);
    lcd.print(angle);

    delay(1000);
  }

  for (int angle = 180; angle >= 0; angle -= 30) {

    myServo.write(angle);

    lcd.setCursor(0, 1);
    lcd.print("    ");
    lcd.setCursor(0, 1);
    lcd.print(angle);

    delay(1000);
  }
}
```
