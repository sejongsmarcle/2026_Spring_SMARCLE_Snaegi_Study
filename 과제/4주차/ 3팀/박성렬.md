#include <LiquidCrystal_I2C.h>
#include <Servo.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

int angles[] = {0, 30, 60, 90, 120, 150, 180};
int numAngles = 7;
int index = 0;

void setup() {
  // 서보모터 초기화
  myServo.attach(9);
  myServo.write(0);

  // LCD 초기화
  lcd.init();
  lcd.backlight();
  lcd.clear();

  

  delay(1000);
}

void loop() {
  int currentAngle = angles[index];

  // 서보모터 각도 설정
  myServo.write(currentAngle);

  // LCD 1행에 각도 출력
  lcd.setCursor(0, 0);
  lcd.print("                "); // 이전 내용 지우기
  lcd.setCursor(0, 0);
  lcd.print(currentAngle);
  lcd.print(" deg");

  index++;
  if (index >= numAngles) {
    index = 0;
  }

  delay(2000);
}

1초 간격으로 서보모터를 30도만큼 회전시키고 회전할 때 마다 lcd에 회전 각도를 출력한다.
