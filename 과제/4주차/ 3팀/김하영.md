#include <Servo.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

void setup() {
  myServo.attach(9);
  lcd.init();
  lcd.backlight();
}

void loop() {
  for (int angle = 0; angle <= 180; angle += 30) {
    myServo.write(angle);
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Servo Angle:");
    lcd.setCursor(0, 1);
    lcd.print(angle);
    delay(1000);


#아두이노가 중심에서 전원을 공급하고 신호를 제어. 브레드보드 전원 레일을 통해 LCD와 서브 모터가
#같은 전원과 접지를 공유함. LCD는 12C 통신을 사용하기에 SDA와 SCL 두 선만으로 아두이노와 데이터를
#주고받을 수 있고, 그 결과 아두이노가 보내는 현재 서보 각도 정보가 LCD에 표시됨. 
#서보 모터는 PWM 신호를 받아 지정된 각도로 움직이고, 아두이노가 디지털 핀을 통해 각도를 순차적으로
#바꿔주면 모터가 실제로 그 위치로 회전한다.
