#include <Servo.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo myServo;

void setup() {
  myServo.attach(9);      
  
  lcd.init();             
  lcd.backlight();        
  
  lcd.setCursor(0, 0);
  lcd.print("Servo System");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(1500);
  lcd.clear();
}

void loop() {
  for (int angle = 0; angle <= 180; angle += 30) {
    
    myServo.write(angle);    
    
    lcd.setCursor(0, 0);
    lcd.print("Current Angle:");
    
    lcd.setCursor(0, 1);
    lcd.print(">> ");
    lcd.print(angle);
    lcd.print(" deg      "); 
    
    delay(1000);          
  }
}

서보모터: 9번 핀에 연결 (PWM 신호)
LCD (I2C): SDA는 A4, SCL은 A5에 연결
전원: 5V랑 GND는 브레드보드 통해서 다 같이 공유
반복문: for문 써서 angle 값을 0부터 180까지 30씩 더하게 만듦. (0, 30, 60... 180)
모터 제어: 계산된 angle 값을 write() 함수로 보내서 모터 돌림.
LCD 출력: 현재 각도를 실시간으로 화면에 찍어줌.
딜레이: 모터가 움직일 시간 필요하니까 1000ms(1초)씩 끊어서 실행.
