#include <LiquidCrystal.h>
#include <Servo.h>

LiquidCrystal lcd(12, 11, 5, 4, 3, 2);
Servo servoMotor;

const int servoPin = 8;

void showAngle(int angle) {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Servo Angle");

  lcd.setCursor(0, 1);
  lcd.print(angle);
  lcd.print((char)223);
}

void moveServo(int angle) {
  servoMotor.write(angle);
  showAngle(angle);
  delay(1000);
}

void setup() {
  lcd.begin(16, 2);
  servoMotor.attach(servoPin);

  lcd.setCursor(0, 0);
  lcd.print("Arduino Servo");
  lcd.setCursor(0, 1);
  lcd.print("LCD Output");

  delay(2000);
  lcd.clear();
}

void loop() {
  for (int angle = 0; angle <= 180; angle += 30) {
    moveServo(angle);
  }

  for (int angle = 180; angle >= 0; angle -= 30) {
    moveServo(angle);
  }
}
