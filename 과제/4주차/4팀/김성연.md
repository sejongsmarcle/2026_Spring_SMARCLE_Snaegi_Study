#include <LiquidCrystal.h>
#include <Servo.h>

// LCD 핀 설정
// RS, E, D4, D5, D6, D7
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

// 서보모터 객체 생성
Servo myServo;

void setup() {

  // LCD 시작
  lcd.begin(16, 2);

  // 서보모터 핀 설정 (D8)
  myServo.attach(8);

  // 시작 화면
  lcd.setCursor(0, 0);
  lcd.print("Servo Control");

  delay(2000);

  lcd.clear();
}

void loop() {

  // 0도 -> 180도
  for(int angle = 0; angle <= 180; angle += 30) {

    // 서보 회전
    if (angle == 0) ;
    else myServo.write(angle);

    // LCD 초기화
    lcd.clear();

    // 첫 번째 줄
    lcd.setCursor(0, 0);
    lcd.print("Current Angle");

    // 두 번째 줄
    lcd.setCursor(0, 1);
    lcd.print(angle);
    lcd.print((char)223); // degree symbol

    delay(1000);
  }

  // 180도 -> 0도
  for(int angle = 180; angle >= 0; angle -= 30) {
    if (angle == 180);

    else myServo.write(angle);

    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("Current Angle");

    lcd.setCursor(0, 1);
    lcd.print(angle);
    lcd.print((char)223);

    delay(1000);
  }
}
