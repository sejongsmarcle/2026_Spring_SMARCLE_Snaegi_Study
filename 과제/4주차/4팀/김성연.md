사용한 하드웨어 : 아두이노 우노, 브레드보드, 서보모터, LCD(16x2) 

코드 내용 요약 : 1. 하드웨어 제어에 필요한 라이브러리 추가 ( servo.h, wire.h, liquidcrystal.h ) 
                2. void setup 함수에서 기본적인 세팅을 함 
                3. void loop 문에서 for문을 이용해 서보 각도를 인자로 해서 각도를 바꾸고 LCD로 각도 출력을 함 
                4. void loop문은 계속 반복되므로 계속 작동하게 됨 
                
시행 착오 과정 : 처음에는 선연결이 적은 i2c통신을 쓰려고 하였지만 코드가 작동을 하지 않았음 -> 고치지 못함 
                그 후에 i2c통신을 버리고 다른 것을 사용함 
                   

#include <LiquidCrystal.h>
#include <Servo.h>

// LCD 핀 설정
// RS, E, D4, D5, D6, D7
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

// 서보모터 객체 생성
Servo myServo;

void setup() {

  // LCD 시작
  lcd.begin(16, 2);

  // 서보모터 핀 설정 (D8)
  myServo.attach(8);

  // 시작 화면
  lcd.setCursor(0, 0);
  lcd.print("Servo Control");

  delay(2000);

  lcd.clear();
}

void loop() {

  // 0도 -> 180도
  for(int angle = 0; angle <= 180; angle += 30) {

    // 서보 회전
    if (angle == 0) ;
    else myServo.write(angle);

    // LCD 초기화
    lcd.clear();

    // 첫 번째 줄
    lcd.setCursor(0, 0);
    lcd.print("Current Angle");

    // 두 번째 줄
    lcd.setCursor(0, 1);
    lcd.print(angle);
    lcd.print((char)223); // degree symbol

    delay(1000);
  }

  // 180도 -> 0도
  for(int angle = 180; angle >= 0; angle -= 30) {
    if (angle == 180);

    else myServo.write(angle);

    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("Current Angle");

    lcd.setCursor(0, 1);
    lcd.print(angle);
    lcd.print((char)223);

    delay(1000);
  }
}
