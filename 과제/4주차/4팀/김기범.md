```cpp
#include <Servo.h>
#include <Adafruit_LiquidCrystal.h>

Servo myServo;

// Tinkercad I2C LCD
Adafruit_LiquidCrystal lcd(0);

const int servoPin = 8;

void setup() {
  myServo.attach(servoPin);

  lcd.begin(16, 2);
  lcd.setBacklight(1);

  lcd.setCursor(0, 0);
  lcd.print("Servo Angle");
}

void loop() {
  for (int angle = 0; angle <= 180; angle += 30) {
    myServo.write(angle);

    lcd.setCursor(0, 0);
    lcd.print("Servo Angle     ");

    lcd.setCursor(0, 1);
    lcd.print("Angle: ");
    lcd.print(angle);
    lcd.print((char)223);  // degree symbol
    lcd.print("      ");

    delay(1000);
  }
}
```
