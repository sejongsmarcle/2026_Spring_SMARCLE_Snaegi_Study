#include <Servo.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// I2C 주소 (보통 0x27 또는 0x3F)
LiquidCrystal_I2C lcd(0x27, 16, 2);

Servo myServo;

int angle = 0;

void setup() {
  myServo.attach(9);   // 서보모터 신호핀 D9

  lcd.init();          // LCD 초기화
  lcd.backlight();     // 백라이트 ON

  lcd.setCursor(0, 0);
  lcd.print("Servo Angle");
}

void loop() {
  for (angle = 0; angle <= 180; angle += 30) {
    
    myServo.write(angle);   // 서보 각도 설정

    lcd.setCursor(0, 1);
    lcd.print("Angle: ");
    lcd.print(angle);
    lcd.print("   "); // 잔상 제거

    delay(1000);
  }
}
