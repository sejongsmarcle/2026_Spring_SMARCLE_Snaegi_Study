```
#include <Servo.h>
#include <LiquidCrystal_I2C.h>

int SERVO_PIN = 1;

Servo myServo;
LiquidCrystal_I2C lcd(0x27, 16, 2);

int currentAngle = 0;

void updateLCD(int angle) {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Servo Angle:");
  lcd.setCursor(0, 1);
  lcd.print(angle);
  lcd.print(" deg");
}

void setup() {
  lcd.init();
  lcd.backlight();
  myServo.attach(SERVO_PIN);
  myServo.write(currentAngle);
  updateLCD(currentAngle);
  delay(600);
}

void loop() {
  for (int angle = 30; angle <= 180; angle += 30) {
    currentAngle = angle;
    myServo.write(currentAngle);
    updateLCD(currentAngle);
    delay(600);
  }
  for (int angle = 150; angle >= 0; angle -= 30) {
    currentAngle = angle;
    myServo.write(currentAngle);
    updateLCD(currentAngle);
    delay(600);
  }
}
```
# 코드
0도부터 시작해서 30도씩 증가하여 180도에 달성하고, 이후 180도부터 다시 0도로 이동하는 코드
- 스터디 중 이용한 라이브러리를 통한 구현
- 서보는 1번 핀에 연결, LCD는 SDA, SCL에 맞는 A4번 A5번 핀 구성
- for문을 통해 angle 값이 30부터 180까지 이동 후, 그 다음 for문으로 150부터 0도까지 이동하는 구조
- 주어진 조건에 따라 LCD 윗줄은 Servo Agnle: 출력, 아래에는 각도 출력
- LCD 출력은 계속해서 반복하므로 함수로 제작

# 회로
- 코드와 동일하게 서보는 1번 핀, LCD는 A4번과 A5번 핀 사용
- 스터디 중 사용한 LiquidCrystal_I2C 라이브러리가 PCF8574 기반 라이브러리이므로 팅커캐드 내 LCD 설정을 해줘야 함
- 동일하게 주소 값 또한 팅커캐드 내 LCD의 값과 일치하게 맞춰야 함
